# Plugin Input/Output

Type definitions for the data structures passed to and returned from plugins.

## CyanPluginInput

The input object passed to the plugin handler.

### Definition

## Full Example

<Tabs items={['TypeScript', 'Python', 'C#']}>
  <Tab value="TypeScript">

  ```ts plugin.ts -c
  import { StartPluginWithLambda, type CyanPluginInput, type PluginOutput } from '@atomicloud/cyan-sdk';
  import { $ } from 'bun';
  import * as fs from 'fs/promises';
  import * as path from 'path';

  // Define your config type
  interface SetupConfig {
    git?: boolean;
    installDeps?: boolean;
    packageManager?: 'npm' | 'bun' | 'yarn' | 'pnpm';
    createReadme?: boolean;
    readmeContent?: string;
  }

  StartPluginWithLambda(async (input) => {
    const { directory } = input;

    // Parse config with defaults
    const config: SetupConfig = {
      git: true,
      installDeps: false,
      packageManager: 'npm',
      createReadme: true,
      ...(input.config as Partial<SetupConfig>)
    };

    // Log input information
    console.log('Plugin Input:');
    console.log('  Directory:', directory);
    console.log('  Config:', JSON.stringify(config, null, 2));

    // Initialize git
    if (config.git) {
      await $`git -C ${directory} init`.quiet();
      await $`git -C ${directory} add .`.quiet();
    }

    // Install dependencies
    if (config.installDeps) {
      const pm = config.packageManager;
      await $`cd ${directory} && ${pm} install`.quiet();
    }

    // Create README
    if (config.createReadme) {
      const readmePath = path.join(directory, 'README.md');
      await fs.writeFile(readmePath, config.readmeContent || '# Project\n');
    }

    // Return output
    return { directory };
  });
  ```

  </Tab>
  <Tab value="Python">

  ```py plugin.py -c
  from atomicloud_cyan_sdk import start_plugin_with_fn, PluginInput, PluginOutput
  import subprocess
  import os
  from dataclasses import dataclass, field
  from typing import Literal, Optional

  @dataclass
  class SetupConfig:
      git: bool = True
      installDeps: bool = False
      packageManager: Literal['npm', 'bun', 'yarn', 'pnpm'] = 'npm'
      createReadme: bool = True
      readmeContent: str = '# Project\n'

  @start_plugin_with_fn
  async def main(input: PluginInput) -> PluginOutput:
      directory = input.directory

      # Parse config with defaults
      defaults = SetupConfig()
      config = SetupConfig(
          git=input.config.get('git', defaults.git),
          installDeps=input.config.get('installDeps', defaults.installDeps),
          packageManager=input.config.get('packageManager', defaults.packageManager),
          createReadme=input.config.get('createReadme', defaults.createReadme),
          readmeContent=input.config.get('readmeContent', defaults.readmeContent)
      )

      # Log input information
      print('Plugin Input:')
      print(f'  Directory: {directory}')
      print(f'  Config: {config}')

      # Initialize git
      if config.git:
          subprocess.run(['git', '-C', directory, 'init'], capture_output=True)
          subprocess.run(['git', '-C', directory, 'add', '.'], capture_output=True)

      # Install dependencies
      if config.installDeps:
          subprocess.run(['cd', directory, '&&', config.packageManager, 'install'], capture_output=True)

      # Create README
      if config.createReadme:
          readme_path = os.path.join(directory, 'README.md')
          with open(readme_path, 'w') as f:
              f.write(config.readmeContent)

      # Return output
      return {'directory': directory}
  ```

  </Tab>
  <Tab value="C#">

  ```csharp Plugin.cs -c
  using Atomicloud.Cyan.Sdk;
  using System.Diagnostics;
  using System.Text.Json;

  public class SetupConfig
  {
      public bool Git { get; set; } = true;
      public bool InstallDeps { get; set; } = false;
      public string PackageManager { get; set; } = "npm";
      public bool CreateReadme { get; set; } = true;
      public string ReadmeContent { get; set; } = "# Project\n";
  }

  [PluginMain]
  public static class Plugin
  {
      public static async Task<PluginOutput> Main(PluginInput input)
      {
          var directory = input.Directory;

          // Parse config with defaults
          var config = new SetupConfig();
          if (input.Config != null)
          {
              var json = JsonDocument.Parse(input.Config.ToString());
              var root = json.RootElement;

              if (root.TryGetProperty("git", out var git))
                  config.Git = git.GetBoolean();
              if (root.TryGetProperty("installDeps", out var installDeps))
                  config.InstallDeps = installDeps.GetBoolean();
              if (root.TryGetProperty("packageManager", out var pm))
                  config.PackageManager = pm.GetString() ?? "npm";
              if (root.TryGetProperty("createReadme", out var createReadme))
                  config.CreateReadme = createReadme.GetBoolean();
              if (root.TryGetProperty("readmeContent", out var readmeContent))
                  config.ReadmeContent = readmeContent.GetString() ?? "# Project\n";
          }

          // Log input information
          Console.WriteLine("Plugin Input:");
          Console.WriteLine($"  Directory: {directory}");
          Console.WriteLine($"  Config: Git={config.Git}, InstallDeps={config.InstallDeps}");

          // Initialize git
          if (config.Git)
          {
              Process.Start("git", $"-C {directory} init")?.WaitForExit();
              Process.Start("git", $"-C {directory} add .")?.WaitForExit();
          }

          // Install dependencies
          if (config.InstallDeps)
          {
              Process.Start(config.PackageManager, $"install --prefix {directory}")?.WaitForExit();
          }

          // Create README
          if (config.CreateReadme)
          {
              var readmePath = Path.Combine(directory, "README.md");
              await File.WriteAllTextAsync(readmePath, config.ReadmeContent);
          }

          // Return output
          return new PluginOutput { Directory = directory };
      }
  }
  ```
