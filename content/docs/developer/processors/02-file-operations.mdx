# File Operations

Processors can perform various file operations beyond simple content transformation. This guide covers advanced file handling techniques.

```ts
export default class MultiFileProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // Read related files
    const packageJson = await input.files.readJson<{
      name: string;
      dependencies: Record<string, string>;
    }>('package.json');

    const tsconfig = await input.files.readJson<{
      compilerOptions: Record<string, any>;
    }>('tsconfig.json');

    // Combine information
    const info = {
      name: packageJson.name,
      hasTypeScript: !!tsconfig,
      dependencyCount: Object.keys(packageJson.dependencies || {}).length
    };

    // Inject info into content
    const content = input.content + '\n\n' + JSON.stringify(info, null, 2);

    return { content, encoding: input.encoding };
  }
}
```

```ts
export default class SideEffectProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // Generate additional files during processing
    const manifest = {
      processedAt: new Date().toISOString(),
      originalPath: input.filePath,
      size: input.content.length
    };

    // Write manifest file
    await input.files.writeJson('.manifest.json', manifest);

}
```

Use `additionalFiles` to generate multiple outputs:

```ts
export default class MultiOutputProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // Parse input (e.g., a component file)
    const componentName = this.extractComponentName(input.content);

    // Generate multiple files
    return {
      content: input.content,
      encoding: input.encoding,
      additionalFiles: [
        {
          path: `${componentName}.test.ts`,
          content: this.generateTest(componentName)
        },
        {
          path: `${componentName}.stories.ts`,
          content: this.generateStory(componentName)
        },
        {
          path: `${componentName}.css`,
          content: this.generateStyles(componentName)
        }
      ]
    };
  }

  private extractComponentName(content: string): string {
    const match = content.match(/export\s+(?:default\s+)?(?:class|function)\s+(\w+)/);
    return match ? match[1] : 'Component';
  }

  private generateTest(name: string): string {
    return `import { render } from '@testing-library/react';
import ${name} from './${name}';

describe('${name}', () => {
  it('renders correctly', () => {
    const { container } = render(<${name} />);
    expect(container).toBeTruthy();
  });
});`;
  }

  private generateStory(name: string): string {
    return `import type { Meta, StoryObj } from '@storybook/react';
import ${name} from './${name}';

const meta: Meta<typeof ${name}> = {
  component: ${name},
};

export default meta;
type Story = StoryObj<typeof ${name}>;

export const Default: Story = {};`;
  }

  private generateStyles(name: string): string {
    return `.${name.toLowerCase()} {
  /* Add styles here */
}`;
  }
}
```

```ts
export default class ConditionalProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // Check if file exists
    const hasConfig = await input.files.exists('custom.config.js');

    let content = input.content;

    if (hasConfig) {
      // Include custom config
      const customConfig = await input.files.read('custom.config.js');
      content = `// Custom config loaded\n${customConfig}\n\n${content}`;
    } else {
      // Use default
      content = `// Using default config\n${content}`;
    }

    return { content, encoding: input.encoding };
  }
}
```

## Directory Operations

<Tabs groupId="sdk-language" items={['TypeScript', '.NET', 'Python']} persist>
  <Tab value="TypeScript">

```ts
export default class DirectoryProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // List directory contents
    const files = await input.files.readdir('src/components');

    // Process each file
    const components = [];
    for (const file of files) {
      if (file.endsWith('.tsx')) {
        const content = await input.files.read(`src/components/${file}`);
        components.push({
          name: file.replace('.tsx', ''),
          exports: this.extractExports(content)
        });
      }
    }

    // Generate index file content
    const indexContent = components
      .map(c => `export { ${c.exports.join(', ')} } from './${c.name}';`)
      .join('\n');

    return {
      content: indexContent,
      encoding: input.encoding
    };
  }

  private extractExports(content: string): string[] {
    const exports: string[] = [];
    const regex = /export\s+(?:const|function|class|interface|type)\s+(\w+)/g;
    let match;
    while ((match = regex.exec(content)) !== null) {
      exports.push(match[1]);
    }
    return exports;
  }
}
```

```csharp
    public async Task<ProcessorOutput> ProcessAsync(CyanProcessorInput input)
    {
        // List directory contents
        var files = await input.Files.ReaddirAsync("src/components");

        // Process each file
        var components = new List<ComponentInfo>();
        foreach (var file in files)
        {
            if (file.EndsWith(".tsx"))

        var indexContent = string.Join("\n",
            components.Select(c => $"export {{ {string.Join(", ", c.Exports)} }} from './{c.Name}';"));

        return new ProcessorOutput

            Content = indexContent,
            Encoding = input.Encoding
        };

    private List<string> ExtractExports(string content)
    {
        var exports = new List<string>();
        var regex = new Regex(@"export\s+(?:const|function|class|interface|type)\s+(\w+)");
        foreach (Match match in regex.Matches(content))
        {
            exports.Add(match.Groups[1].Value);
        }
        return exports;
    }
```

```python
import re

class DirectoryProcessor(ICyanProcessor):
    async def process(self, input: CyanProcessorInput) -> ProcessorOutput:
        # List directory contents
        files = await input.files.readdir("src/components")

        # Process each file
        components = []
        for file in files:
            if file.endswith(".tsx"):
                content = await input.files.read(f"src/components/{file}")
                components.append({
                    "name": file.replace(".tsx", ""),
                    "exports": self._extract_exports(content)

        index_lines = [
            f"export {{ {', '.join(c['exports'])} }} from './{c['name']}';"

        pattern = r'export\s+(?:const|function|class|interface|type)\s+(\w+)'
        for match in re.finditer(pattern, content):
            exports.append(match.group(1))
        return exports
```

## Skip File Processing

Return with `skipped: true` to skip output:

<Tabs groupId="sdk-language" items={['TypeScript', '.NET', 'Python']} persist>

```ts
export default class FilteringProcessor implements ICyanProcessor {
  async process(input: CyanProcessorInput): Promise<ProcessorOutput> {
    // Skip certain files
    if (input.filePath.includes('.test.')) {
      return {
        content: '',
        encoding: input.encoding,
        skipped: true,
        skipReason: 'Test files are not processed'
      };
    }

    // Skip binary files
    if (this.isBinary(input.content)) {
      return {
        content: input.content,
        encoding: input.encoding,
        skipped: true,
        skipReason: 'Binary files are skipped'
      };
    }

    // Process normally
    return {
      content: this.transform(input.content),
      encoding: input.encoding
    };
  }

  private isBinary(content: string): boolean {
    // Simple binary detection
    return /[\x00-\x08\x0E-\x1F]/.test(content);
  }

  private transform(content: string): string {
    return content;
  }
}
```
