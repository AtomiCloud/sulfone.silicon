# Access Processor Config

Processors receive configuration from the template through `input.config`. This allows templates to customize processor behavior.

## ProcessorConfig Type

<TypeTable
  type={{
    customSetting: {
      type: 'boolean',
      description: 'Enable custom behavior',
    },
    features: {
      type: 'string[]',
      description: 'List of enabled features',
    },
    outputFormat: {
      type: "'json' | 'yaml'",
      description: 'Output format for generated files',
    },
  }}
/>

## Template Configuration

In the template's `index.ts`, pass config to the processor:

```ts index.ts -c
// Template's index.ts
import { StartTemplateWithLambda, GlobType } from '@atomicloud/cyan-sdk';

interface ProcessorConfig {
  customSetting: boolean;
  features: string[];
  outputFormat: 'json' | 'yaml';
}

StartTemplateWithLambda(async (i, d) => {
  // Ask user for preferences
  const features = await i.checkbox(
    'Select features:',
    ['docker', 'ci', 'testing'],
    'project.features',
    'Choose features'
  );

  const format = await i.select(
    'Output format:',
    ['json', 'yaml'],
    'config.format',
    'Choose format'
  );

  return {
    processors: [{
      name: 'myorg/my-processor',
      files: [{ root: 'templates', glob: '**/*', exclude: [], type: GlobType.Template }],
      config: {
        customSetting: true,
        features: features,
        outputFormat: format
      } as ProcessorConfig
    }]
  };
});
```

## Accessing Config in Processor

```ts index.ts -c
// Processor's index.ts
import { StartProcessorWithLambda } from '@atomicloud/cyan-sdk';

interface ProcessorConfig {
  customSetting: boolean;
  features: string[];
  outputFormat: 'json' | 'yaml';
}

StartProcessorWithLambda(async (input, fileHelper) => {
  // Cast config to your type
  const config = input.config as ProcessorConfig;

  // Use config values
  if (config.customSetting) {
    console.log('Custom setting enabled');
  }

  const files = fileHelper.resolveAll();

  files.forEach(file => {
    if (config.features.includes('docker')) {
      // Modify file.content as needed for docker feature
    }

    if (config.outputFormat === 'yaml') {
      // Convert file.content to YAML format
    }

    file.writeFile();
  });

  return { directory: input.writeDir };
});
```

## Default Config Values

Always handle missing config gracefully:

```ts index.ts -c
interface ProcessorConfig {
  includeComments?: boolean;
  indentSize?: number;
  generateTypes?: boolean;
}

StartProcessorWithLambda(async (input, fileHelper) => {
  // Provide defaults for missing values
  const config: ProcessorConfig = {
    includeComments: true,
    indentSize: 2,
    generateTypes: false,
    ...(input.config as Partial<ProcessorConfig>)
  };

  const files = fileHelper.resolveAll();

  files.forEach(file => {
    const indent = ' '.repeat(config.indentSize);

    if (config.generateTypes) {
      // Generate types from file.content
    }

    if (!config.includeComments) {
      // Remove comments from file.content
    }

    file.writeFile();
  });

  return { directory: input.writeDir };
});
```

## Complex Config Example

```ts template-config.ts -c
// Template config
{
  name: 'myorg/codegen-processor',
  files: [{ root: 'schemas', glob: '**/*.graphql', exclude: [], type: GlobType.Template }],
  config: {
    generateClient: true,
    generateServer: false,
    clientOptions: {
      outputDir: 'src/client',
      fetcher: 'axios',
      typesOnly: false
    },
    scalars: {
      DateTime: 'Date',
      JSON: 'Record<string, unknown>'
    }
  }
}

// Processor handling complex config
interface CodegenConfig {
  generateClient?: boolean;
  generateServer?: boolean;
  clientOptions?: {
    outputDir: string;
    fetcher: 'axios' | 'fetch';
    typesOnly: boolean;
  };
  scalars?: Record<string, string>;
}

StartProcessorWithLambda(async (input, fileHelper) => {
  const config: CodegenConfig = input.config as CodegenConfig;

  const files = fileHelper.resolveAll();

  files.forEach(file => {
    // Parse schema from file.content
    const schema = parseSchema(file.content);

    if (config.generateClient) {
      const clientCode = generateClient(schema, config.clientOptions);
      // Note: To write additional files, use fileHelper to create new VirtualFile instances
      // and write them to the appropriate output location within input.writeDir
    }

    file.writeFile();
  });

  return { directory: input.writeDir };
});
```

<Callout type="info">
Config is passed as `unknown` type. Always cast to your expected type and provide default values for optional fields.
</Callout>
