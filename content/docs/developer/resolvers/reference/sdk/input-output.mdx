## Usage Example

<Tabs items={['TypeScript', 'Python', 'C#']}>
  <Tab value="TypeScript">

```ts index.ts -c
import { ResolverOutput, StartResolverWithLambda } from '@atomicloud/cyan-sdk';

StartResolverWithLambda(async (input): Promise<ResolverOutput> => {
  // Access config
  const strategy = input.config['arrayStrategy'] as string || 'replace';

  // Sort files by layer (lower = higher priority)
  const sorted = [...input.files].sort(
    (a, b) => a.origin.layer - b.origin.layer
  );

  // Merge content
  let merged = {};
  for (const file of sorted) {
    const content = JSON.parse(file.content);
    merged = { ...merged, ...content };
  }

  return {
    path: input.files[0].path,
    content: JSON.stringify(merged, null, 2)
  };
});
```

  </Tab>
  <Tab value="Python">

```python main.py -c
import json
from typing import Any
from cyanprintsdk.domain.resolver.input import ResolverInput, ResolvedFile
from cyanprintsdk.domain.resolver.output import ResolverOutput
from cyanprintsdk.main import start_resolver_with_fn


async def resolver(input: ResolverInput) -> ResolverOutput:
    # Access config
    strategy: str = input.config.get('arrayStrategy', 'replace')

    # Sort files by layer (lower = higher priority)
    sorted_files = sorted(input.files, key=lambda f: f.origin.layer)

    # Merge content
    merged: dict[str, Any] = {}
    for file in sorted_files:
        content = json.loads(file.content)
        merged = {**merged, **content}

    return ResolverOutput(
        path=input.files[0].path,
        content=json.dumps(merged, indent=2)
    )


if __name__ == "__main__":
    start_resolver_with_fn(resolver)
```

  <Tab value="C#">

```cs Program.cs -c
using System.Text.Json;
using sulfone_helium;
using sulfone_helium.Domain.Resolver;

CyanEngine.StartResolver(
    args,
    async (ResolverInput input) =>
    {
        // Access config
        var strategy = input.Config.TryGetValue("arrayStrategy", out var stratObj)
            ? stratObj?.ToString() ?? "replace"
            : "replace";

        // Sort files by layer (lower = higher priority)
        var sorted = input.Files.OrderBy(f => f.Origin.Layer).ToList();

        // Merge content
        var merged = new Dictionary<string, object>();
        foreach (var file in sorted)
        {
            var content = JsonSerializer.Deserialize<Dictionary<string, object>>(file.Content)
                ?? new Dictionary<string, object>();
            foreach (var kvp in content)
            {
                merged[kvp.Key] = kvp.Value;
            }
        }

        return new ResolverOutput(
            input.Files.First().Path,
            JsonSerializer.Serialize(merged, new JsonSerializerOptions { WriteIndented = true })
        );
    }
);
```
