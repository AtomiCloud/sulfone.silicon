# .NET SDK Reference

Complete API reference for the `AtomiCloud.Helium` package.

## Installation

```bash
dotnet add package AtomiCloud.Helium
```

## Namespaces

```csharp
using AtomiCloud.Helium;
using AtomiCloud.Helium.Processors;
using AtomiCloud.Helium.Plugins;
using AtomiCloud.Helium.Inquirer;
```

---

## Core Interfaces

### ICyanTemplate

All templates implement this interface:

```csharp
public interface ICyanTemplate
{
    Task ExecuteAsync(Cyan cyan);
}
```

### ICyanProcessor

All processors implement this interface:

```csharp
public interface ICyanProcessor
{
    Task<ProcessorOutput> ProcessAsync(CyanProcessorInput input);
}
```

### ICyanPlugin

All plugins implement this interface:

```csharp
public interface ICyanPlugin
{
    string Name { get; }
    string Version { get; }
    Task InitializeAsync(PluginContext context);
    Task DisposeAsync();
}
```

## Cyan Object

The main object passed to templates.

```csharp
public class Cyan
{
    // Properties
    public Answers Answers { get; }
    public Dictionary<string, object> Config { get; }
    public Inquirer Inquirer { get; }
    public Determinism Determinism { get; }
    public FileHelper Files { get; }
    public ProcessorRunner Processors { get; }
    public PluginManager Plugins { get; }
    public ChildRunner Children { get; }
    public Logger Logger { get; }

    // Methods
    public string[] Glob(string pattern);
    public string[] Glob(string[] patterns);
    public string[] Glob(string pattern, GlobOptions options);
}

public class GlobOptions
{
    public string Cwd { get; set; }
    public bool Dot { get; set; }
    public bool Absolute { get; set; }
    public bool OnlyFiles { get; set; }
    public bool OnlyDirectories { get; set; }
}
```

## Answers

Access to user answers.

```csharp
public class Answers
{
    // Indexer
    public object this[string key] { get; set; }

    // Methods
    public T Get<T>(string key);
    public T GetOrDefault<T>(string key, T defaultValue);
    public bool Has(string key);
    public void Set(string key, object value);
    public bool TryGetValue<T>(string key, out T value);

    // Conditional prompting
    public Task<string> GetOrPromptAsync(string key, TextOptions options);
    public Task<string> GetOrPromptSelectAsync(string key, SelectOptions options);
    public Task<string[]> GetOrPromptCheckboxAsync(string key, CheckboxOptions options);
}
```

## Inquirer

User interaction methods.

```csharp
public class Inquirer
{
    // Text input
    public Task<string> TextAsync(TextOptions options);

    // Single selection
    public Task<string> SelectAsync(SelectOptions options);

    // Multiple selections
    public Task<string[]> CheckboxAsync(CheckboxOptions options);

    // Yes/No confirmation
    public Task<bool> ConfirmAsync(ConfirmOptions options);

    // Masked input
    public Task<string> PasswordAsync(PasswordOptions options);

    // Date picker
    public Task<DateTime> DateAsync(DateOptions options);

    // Number input
    public Task<int> NumberAsync(NumberOptions options);

    // Expandable selection
    public Task<object> ExpandAsync(ExpandOptions options);
}

// Option classes
public class TextOptions
{
    public string Message { get; set; }
    public string Default { get; set; }
    public string Placeholder { get; set; }
    public Func<string, string> Validate { get; set; }
    public Func<string, string> Transformer { get; set; }
    public bool Required { get; set; }
}

public class SelectOption
{
    public string Value { get; set; }
    public string Label { get; set; }
    public string Description { get; set; }
    public bool Disabled { get; set; }
}

public class SelectOptions
{
    public string Message { get; set; }
    public object[] Options { get; set; }
    public string Default { get; set; }
    public int PageSize { get; set; }
}

public class CheckboxOption
{
    public string Value { get; set; }
    public string Label { get; set; }
    public bool Checked { get; set; }
    public bool Disabled { get; set; }
}

public class CheckboxOptions
{
    public string Message { get; set; }
    public object[] Options { get; set; }
    public bool Required { get; set; }
    public int Min { get; set; }
    public int Max { get; set; }
    public int PageSize { get; set; }
}

public class ConfirmOptions
{
    public string Message { get; set; }
    public bool Default { get; set; }
}

public class PasswordOptions
{
    public string Message { get; set; }
    public string Mask { get; set; }
    public Func<string, string> Validate { get; set; }
}

public class DateOptions
{
    public string Message { get; set; }
    public DateTime? Default { get; set; }
    public DateTime? Min { get; set; }
    public DateTime? Max { get; set; }
    public string Format { get; set; }
}

public class NumberOptions
{
    public string Message { get; set; }
    public int? Default { get; set; }
    public int? Min { get; set; }
    public int? Max { get; set; }
    public int? Step { get; set; }
    public Func<int, string> Validate { get; set; }
}

public class ExpandOption
{
    public string Key { get; set; }
    public string Name { get; set; }
    public object Value { get; set; }
}

public class ExpandOptions
{
    public string Message { get; set; }
    public ExpandOption[] Options { get; set; }
    public string Default { get; set; }
}
```

### HookRegistry

```csharp
public class HookRegistry
{
    public void BeforeExecute(Func<BeforeExecuteEvent, Task> callback);
    public void AfterInquirer(Func<AfterInquirerEvent, Task> callback);
    public void BeforeProcess(Func<BeforeProcessEvent, Task> callback);
    public void AfterProcess(Func<AfterProcessEvent, Task> callback);
```

### Hook Event Types

```csharp
public class BeforeExecuteEvent
{
    public string TemplateName { get; set; }
    public string TemplateVersion { get; set; }
    public Dictionary<string, object> Answers { get; set; }
    public DateTime Timestamp { get; set; }

{

public class AfterProcessEvent
{
    public string TemplateName { get; set; }

public class AfterExecuteEvent

}
```
