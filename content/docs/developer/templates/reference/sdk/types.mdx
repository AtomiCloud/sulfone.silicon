# Type Definitions

Complete type definitions for the CyanPrint SDK.

## Enums

### GlobType

File processing types.

<TypeTable
  type={{
    Template: {
      type: '0',
      description: 'Process with variable substitution (e.g., {{projectName}})',
    },
    Copy: {
      type: '1',
      description: 'Copy without any processing (e.g., images, binaries)',
    },
  }}
/>

**Usage:**

```ts cyan.config.ts -c
import { GlobType } from '@atomicloud/cyan-sdk';

files: [
  { root: 'templates', glob: '**/*.md', exclude: [], type: GlobType.Template },
  { root: 'templates', glob: '**/*.png', exclude: [], type: GlobType.Copy }
]
```

  desc: 'Enter project name'
});

### IInquirer

Question asking interface.

```ts -c
interface IInquirer {
  // Shorthand forms
  text(q: string, id: string, help?: string | null): Promise<string>;
  select(q: string, options: string[], id: string, help?: string | null): Promise<string>;
  confirm(q: string, id: string, help?: string | null): Promise<boolean>;
  checkbox(q: string, options: string[], id: string, help?: string | null): Promise<string[]>;
  password(q: string, id: string, help?: string | null): Promise<string>;
  dateSelect(q: string, id: string, help?: string | null): Promise<string>;

  // Object forms
  text(config: TextQ): Promise<string>;
  select(config: SelectQ): Promise<string>;
  confirm(config: ConfirmQ): Promise<boolean>;
  checkbox(config: CheckboxQ): Promise<string[]>;
  password(config: PasswordQ): Promise<string>;
  dateSelect(config: DateQ): Promise<string>;
}
```
