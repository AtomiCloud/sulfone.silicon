# Type Definitions

Complete type definitions for the CyanPrint SDK.

## Enums

### GlobType

File processing types.

```ts
enum GlobType {
  Template = 0,  // Process with variable substitution
  Copy = 1,      // Copy without processing
  Ignore = 2     // Skip entirely
}
```

**Usage:**

```ts
import { GlobType } from '@atomicloud/cyan-sdk';

files: [
  { root: 'templates', glob: '**/*.md', exclude: [], type: GlobType.Template },
  { root: 'templates', glob: '**/*.png', exclude: [], type: GlobType.Copy }
]
```

---

### QuestionType

Question type identifiers.

```ts
enum QuestionType {
  Text = 0,
  DateSelect = 1,
  Select = 2,
  Checkbox = 3,
  Password = 4,
  Confirm = 5
}
```

**Usage:**

```ts
import { QuestionType } from '@atomicloud/cyan-sdk';

const name = await i.text({
  type: QuestionType.Text,
  id: 'project.name',
  message: 'Project name?',
  desc: 'Enter project name'
});
```

### IFileGroup

File group for processing.

```ts
interface IFileGroup {
  root: string;       // Source directory
  glob: string;       // Glob pattern
  exclude: string[];  // Exclude patterns
  type: GlobType;     // Processing type
}
```

### IInquirer

Question asking interface.

```ts
interface IInquirer {
  // Shorthand forms
  text(message: string, id: string, desc?: string): Promise<string>;
  select(message: string, options: string[], id: string, desc?: string): Promise<string>;
  confirm(message: string, id: string, desc?: string): Promise<boolean>;
  checkbox(message: string, options: string[], id: string, desc?: string): Promise<string[]>;
  password(message: string, id: string, desc?: string): Promise<string>;
  dateSelect(message: string, id: string, desc?: string): Promise<Date>;

  // Object forms
  text(config: ITextQuestion): Promise<string>;
  select(config: ISelectQuestion): Promise<string>;
  confirm(config: IConfirmQuestion): Promise<boolean>;
  checkbox(config: ICheckboxQuestion): Promise<string[]>;
  password(config: IPasswordQuestion): Promise<string>;
  dateSelect(config: IDateQuestion): Promise<Date>;
}
```

## Type Aliases

### ValidateFunction

```ts
type ValidateFunction<T> = (input: T) => string | null;
```

Returns `null` for valid input, or an error message string for invalid input.
